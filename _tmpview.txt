using IndQuestResults.Performance;

namespace IndQuestResults.Tests.Unit.Performance;

public class MemoryOptimizationsTheoryTests
{
    [Theory]
    [MemberData(nameof(CombineArraysCases))]
    public void CombineArrays_Various(params int[][]? arrays)
    {
        var combined = MemoryOptimizations.CombineArrays(arrays!);
        var expected = arrays is null || arrays.Length == 0
            ? Array.Empty<int>()
            : arrays.Where(a => a is { Length: > 0 }).SelectMany(a => a).ToArray();
        combined.ShouldBe(expected);
    }

    public static TheoryData<int[][]?> CombineArraysCases()
    {
        var data = new TheoryData<int[][]?>();
        data.Add(null);
        data.Add(Array.Empty<int[]>());
        data.Add([Array.Empty<int>()]);
        data.Add([new[] { 1, 2 }]);
        data.Add([new[] { 1 }, Array.Empty<int>(), new[] { 2, 3 }]);
        return data;
    }

    [Theory]
    [InlineData(new int[] { })]
    [InlineData(new int[] { 1, 2, 3 })]
    public void ToArrayOptimized_CollectionAndArray(params int[] input)
    {
        var asArray = MemoryOptimizations.ToArrayOptimized(input);
        asArray.ShouldBe(input);

        var asList = MemoryOptimizations.ToArrayOptimized((IEnumerable<int>)input.ToList());
        asList.ShouldBe(input);
    }
}
